cmake_minimum_required(VERSION 3.25)
project(PhoenixV3 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
include(FetchContent)
set(FETCHCONTENT_QUIET OFF)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FLECS_SHARED OFF CACHE BOOL "" FORCE)
set(FLECS_STATIC ON CACHE BOOL "" FORCE)
FetchContent_Declare(flecs URL https://codeload.github.com/SanderMertens/flecs/tar.gz/d7d0c4f7afb4518a6bae749efdc52c7cb5cffee6 URL_HASH SHA256=F0030372BCB82F5601D08A489945D55409074C4430B9D2C2983C82492E3F8E85 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(raylib URL https://codeload.github.com/raysan5/raylib/tar.gz/dbc56a87da87d973a9c5baa4e7438a9d20121d28 URL_HASH SHA256=81B06CE7C19CF3B634B0271C23C361BA6AD8BF45FB8B036ABBFEB4260EC1E126 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
FetchContent_Declare(freetype URL https://codeload.github.com/freetype/freetype/tar.gz/534ad3456055ee1f65ecde3bcf22a656a31514d1 URL_HASH SHA256=63FCF92CB44EFA3B169207BFF068676CD8EA188CCECC893812B1877AD36855C5 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(flecs raylib freetype)
add_library(Freetype::Freetype ALIAS freetype)
set(FREETYPE_INCLUDE_DIRS "${freetype_SOURCE_DIR}/include" CACHE STRING "" FORCE)
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(rmlui URL https://codeload.github.com/mikke89/RmlUi/tar.gz/613b206bb313808fc45924b3e6ecfbcf35b91438 URL_HASH SHA256=9B2D208E66AC4C7E8382664599DBE66F9BDA35D8739ABF41D94C3223F81CE663 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/tar.gz/8a223b61bd671687b254458333bb1e3fad50ed5f URL_HASH SHA256=AF1EFF67CB527AA80B5F9B48249E291C5AFF2E9E69C39B2F30276CD1F7B9A200 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(rmlui imgui)
add_library(phoenix_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp)
target_include_directories(phoenix_imgui PUBLIC ${imgui_SOURCE_DIR})
add_library(phoenix_core STATIC src/core/application.cpp src/core/semantics.cpp src/core/input.cpp)
target_include_directories(phoenix_core PUBLIC include)
add_library(phoenix_simulation STATIC src/simulation/world.cpp)
target_link_libraries(phoenix_simulation PUBLIC phoenix_core flecs_static)
add_library(phoenix_presentation STATIC src/presentation/screen.cpp)
if(EMSCRIPTEN)
  target_sources(phoenix_presentation PRIVATE src/presentation/browser_semantics.cpp)
else()
  enable_testing()
  add_executable(phoenix_system_tests tests/system/input_and_semantics.cpp)
  target_link_libraries(phoenix_system_tests PRIVATE phoenix_core)
  add_test(NAME system.input-and-semantics COMMAND phoenix_system_tests)
  target_sources(phoenix_presentation PRIVATE src/presentation/windows_semantics.cpp)
  target_link_libraries(phoenix_presentation PRIVATE uiautomationcore ole32 oleaut32 comctl32)
endif()
target_link_libraries(phoenix_presentation PUBLIC phoenix_core raylib RmlUi::RmlUi)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/assets")
configure_file("${rmlui_SOURCE_DIR}/Samples/assets/LatoLatin-Regular.ttf" "${CMAKE_CURRENT_BINARY_DIR}/assets/LatoLatin-Regular.ttf" COPYONLY)
configure_file("${rmlui_SOURCE_DIR}/Samples/assets/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/assets/FONT-LICENSE.txt" COPYONLY)
add_library(phoenix_tools STATIC src/tools/workshop.cpp)
target_link_libraries(phoenix_tools PUBLIC phoenix_presentation phoenix_imgui)
if(EMSCRIPTEN)
  add_executable(phoenix_build_check src/entry/build_check.cpp)
  target_link_libraries(phoenix_build_check PRIVATE phoenix_simulation)
  target_link_options(phoenix_build_check PRIVATE "-sALLOW_MEMORY_GROWTH=1" "-sASSERTIONS=2" "-sSTACK_SIZE=1048576" "-sSTACK_OVERFLOW_CHECK=2" "-g2")
  foreach(role IN ITEMS server client workshop)
    add_executable(${role} src/entry/web.cpp)
    target_compile_definitions(${role} PRIVATE PHOENIX_ROLE="${role}")
    target_link_libraries(${role} PRIVATE phoenix_presentation)
    target_link_options(${role} PRIVATE "-sALLOW_MEMORY_GROWTH=1" "-sASSERTIONS=1" "-sFORCE_FILESYSTEM=1" "-sSTACK_SIZE=1048576" "-sSTACK_OVERFLOW_CHECK=2" "-g2")
    target_link_options(${role} PRIVATE "--preload-file" "${CMAKE_CURRENT_BINARY_DIR}/assets@/assets")
    target_link_options(${role} PRIVATE "-sEXPORTED_RUNTIME_METHODS=['ccall']")
  endforeach()
  target_link_libraries(server PRIVATE phoenix_simulation)
  target_compile_definitions(server PRIVATE PHOENIX_SERVER=1)
  target_link_libraries(workshop PRIVATE phoenix_tools)
  target_compile_definitions(workshop PRIVATE PHOENIX_WORKSHOP=1)
  configure_file(web/index.html "${CMAKE_CURRENT_BINARY_DIR}/index.html" COPYONLY)
else()
  add_executable(native src/entry/native.cpp)
  target_link_libraries(native PRIVATE phoenix_simulation phoenix_tools)
  add_custom_command(TARGET native POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/assets" "$<TARGET_FILE_DIR:native>/assets")
endif()
