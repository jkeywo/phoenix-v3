add_library(phoenix_files STATIC imports.cpp)
target_sources(phoenix_files PUBLIC FILE_SET CXX_MODULES FILES files.cppm)
target_link_libraries(phoenix_files PUBLIC phoenix_foundation)
set(phoenix_file_dependencies phoenix_foundation)
if(EMSCRIPTEN)
	target_link_libraries(phoenix_files PRIVATE phoenix_content_packs)
	list(APPEND phoenix_file_dependencies phoenix_content_packs)
	target_sources(phoenix_files PRIVATE browser_files.cpp)
elseif(WIN32)
	target_sources(phoenix_files PRIVATE windows_files.cpp)
	target_link_libraries(phoenix_files PRIVATE comdlg32)
else()
	find_package(PkgConfig REQUIRED)
	pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET GLOBAL gtk+-3.0)
	target_sources(phoenix_files PRIVATE linux_files.cpp)
	target_link_libraries(phoenix_files PRIVATE PkgConfig::GTK3)
	list(APPEND phoenix_file_dependencies PkgConfig::GTK3)
endif()
set_property(TARGET phoenix_files PROPERTY CXX_SCAN_FOR_MODULES ON)
set_property(TARGET phoenix_files PROPERTY PHOENIX_ALLOWED_DEPENDENCIES "${phoenix_file_dependencies}")
set_property(GLOBAL APPEND PROPERTY PHOENIX_DEPENDENCY_MODULES phoenix_files)

# User directory discovery has no file-picker, content or application dependency.
add_library(phoenix_platform_user_paths STATIC user_paths.cpp)
target_sources(phoenix_platform_user_paths PUBLIC FILE_SET CXX_MODULES FILES user_paths.cppm)
set_target_properties(
	phoenix_platform_user_paths
	PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_SCAN_FOR_MODULES ON
)
if(WIN32)
	target_link_libraries(phoenix_platform_user_paths PRIVATE shell32 ole32)
	set_property(TARGET phoenix_platform_user_paths PROPERTY PHOENIX_ALLOWED_DEPENDENCIES "shell32;ole32")
else()
	set_property(TARGET phoenix_platform_user_paths PROPERTY PHOENIX_ALLOWED_DEPENDENCIES "")
endif()
set_property(GLOBAL APPEND PROPERTY PHOENIX_DEPENDENCY_MODULES phoenix_platform_user_paths)
