add_library(phoenix_platform_lifecycle STATIC)
target_sources(phoenix_platform_lifecycle PUBLIC FILE_SET CXX_MODULES FILES lifecycle.cppm)
target_link_libraries(phoenix_platform_lifecycle PUBLIC phoenix_foundation)
set_property(TARGET phoenix_platform_lifecycle PROPERTY CXX_SCAN_FOR_MODULES ON)
if(EMSCRIPTEN)
	target_compile_options(phoenix_platform_lifecycle PUBLIC ${PHOENIX_EXCEPTION_FLAG})
	target_link_options(phoenix_platform_lifecycle INTERFACE ${PHOENIX_EXCEPTION_FLAG})
endif()
set_property(TARGET phoenix_platform_lifecycle PROPERTY PHOENIX_ALLOWED_DEPENDENCIES "phoenix_foundation")
set_property(GLOBAL APPEND PROPERTY PHOENIX_DEPENDENCY_MODULES phoenix_platform_lifecycle)
