# Byte blobs and peer journals have separate named interfaces and runtime targets.
add_library(phoenix_blobs STATIC blob.cpp)
target_sources(phoenix_blobs PUBLIC FILE_SET CXX_MODULES FILES blobs.cppm)
target_link_libraries(phoenix_blobs PUBLIC phoenix_foundation)
set(phoenix_blob_dependencies phoenix_foundation)

add_library(phoenix_platform_storage STATIC journal.cpp)
target_sources(phoenix_platform_storage PUBLIC FILE_SET CXX_MODULES FILES storage.cppm)
target_link_libraries(phoenix_platform_storage PUBLIC phoenix_replication)
set(phoenix_journal_dependencies phoenix_replication)
if(EMSCRIPTEN)
	target_sources(phoenix_blobs PRIVATE browser_blobs.cpp)
	target_sources(phoenix_platform_storage PRIVATE browser_store.cpp)
	target_link_libraries(phoenix_blobs PRIVATE phoenix_platform_lifecycle)
	target_link_libraries(phoenix_platform_storage PRIVATE phoenix_platform_lifecycle)
	list(APPEND phoenix_blob_dependencies phoenix_platform_lifecycle)
	list(APPEND phoenix_journal_dependencies phoenix_platform_lifecycle)
else()
	target_sources(phoenix_blobs PRIVATE native_blobs.cpp)
	target_sources(phoenix_platform_storage PRIVATE native_store.cpp)
	if(WIN32)
		target_sources(phoenix_platform_storage PRIVATE windows_store.cpp)
	else()
		target_sources(phoenix_platform_storage PRIVATE posix_store.cpp)
	endif()
endif()
foreach(target phoenix_blobs phoenix_platform_storage)
	set_property(TARGET ${target} PROPERTY CXX_SCAN_FOR_MODULES ON)
	set_property(GLOBAL APPEND PROPERTY PHOENIX_DEPENDENCY_MODULES ${target})
endforeach()
set_property(TARGET phoenix_blobs PROPERTY PHOENIX_ALLOWED_DEPENDENCIES "${phoenix_blob_dependencies}")
set_property(TARGET phoenix_platform_storage PROPERTY PHOENIX_ALLOWED_DEPENDENCIES "${phoenix_journal_dependencies}")
