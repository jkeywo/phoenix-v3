add_library(phoenix_transport STATIC options.cpp policy.cpp content_download.cpp)
target_sources(phoenix_transport PUBLIC FILE_SET CXX_MODULES FILES transport.cppm)
target_link_libraries(phoenix_transport PUBLIC phoenix_foundation PRIVATE nlohmann_json::nlohmann_json)
set(phoenix_transport_dependencies phoenix_foundation nlohmann_json::nlohmann_json)
if(EMSCRIPTEN)
	target_sources(phoenix_transport PRIVATE browser_transport.cpp)
	target_link_libraries(phoenix_transport PRIVATE phoenix_platform_lifecycle)
	target_link_options(phoenix_transport PUBLIC "-sFETCH=1")
	list(APPEND phoenix_transport_dependencies phoenix_platform_lifecycle)
else()
	target_sources(phoenix_transport PRIVATE native_transport.cpp)
	target_link_libraries(phoenix_transport PRIVATE datachannel-static)
	list(APPEND phoenix_transport_dependencies datachannel-static)
	if(WIN32)
		target_link_libraries(phoenix_transport PRIVATE crypt32 winhttp)
	else()
		find_package(CURL REQUIRED GLOBAL)
		target_link_libraries(phoenix_transport PRIVATE CURL::libcurl)
		list(APPEND phoenix_transport_dependencies CURL::libcurl)
	endif()
endif()
set_property(TARGET phoenix_transport PROPERTY CXX_SCAN_FOR_MODULES ON)
set_property(TARGET phoenix_transport PROPERTY PHOENIX_ALLOWED_DEPENDENCIES "${phoenix_transport_dependencies}")
set_property(GLOBAL APPEND PROPERTY PHOENIX_DEPENDENCY_MODULES phoenix_transport)
