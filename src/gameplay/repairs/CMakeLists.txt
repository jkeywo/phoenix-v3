add_library(phoenix_gameplay_repairs STATIC repairs.cpp)
target_sources(phoenix_gameplay_repairs PUBLIC FILE_SET CXX_MODULES FILES repairs.cppm)
target_link_libraries(
	phoenix_gameplay_repairs
	PUBLIC phoenix_gameplay_combat_state
	PRIVATE phoenix_gameplay_combat_faults
)
set_target_properties(
	phoenix_gameplay_repairs
	PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_SCAN_FOR_MODULES ON
)
set_property(
	TARGET phoenix_gameplay_repairs
	PROPERTY PHOENIX_ALLOWED_DEPENDENCIES "phoenix_gameplay_combat_state;phoenix_gameplay_combat_faults"
)
set_property(GLOBAL APPEND PROPERTY PHOENIX_DEPENDENCY_MODULES phoenix_gameplay_repairs)
